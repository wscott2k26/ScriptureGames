name: RC3 Certification V3

on:
  workflow_dispatch:
  push:
    branches: [main]
    paths:
      - '.github/workflows/rc3-certification-v3.yml'

permissions:
  contents: write
  issues: write

env:
  RC3_SOURCE_BRANCH: rc3-upload
  RC3_ZIP: ScriptureGames-Cinematic-Visual-Master-RC3.zip
  RC3_SHA256: 0ac177ef5ff9b408a140ffb623295ae00c09171c366e587706d0b6d81c5592b2
  EAS_PROJECT_ID: 18710e80-2a62-46ac-8d3a-d1711b7d920a
  EAS_OWNER: wscott2k8

jobs:
  certify-and-publish:
    runs-on: ubuntu-latest
    timeout-minutes: 240
    env:
      GH_TOKEN: ${{ github.token }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: main
          fetch-depth: 0

      - uses: actions/checkout@v4
        with:
          ref: ${{ env.RC3_SOURCE_BRANCH }}
          path: rc3-upload-source
          fetch-depth: 1

      - uses: actions/setup-node@v4
        with:
          node-version: '20'

      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Verify and extract the exact RC3 package
        shell: bash
        run: |
          set -euo pipefail
          zip_path="$GITHUB_WORKSPACE/rc3-upload-source/$RC3_ZIP"
          test -f "$zip_path"
          printf '%s  %s\n' "$RC3_SHA256" "$zip_path" | sha256sum --check --strict
          rm -rf /tmp/sg-rc3 && mkdir -p /tmp/sg-rc3
          unzip -q "$zip_path" -d /tmp/sg-rc3
          source_root="/tmp/sg-rc3/ScriptureGames-Cinematic-Visual-Master-RC3"
          test -f "$source_root/frontend/package.json"
          test -f "$source_root/scripts/audit.py"
          test -f "$source_root/scripts/audit-visual-master.py"
          echo "SOURCE_ROOT=$source_root" >> "$GITHUB_ENV"

      - name: Apply verified release corrections
        shell: bash
        run: |
          set -euo pipefail
          python - <<'PY'
          import json
          import os
          from pathlib import Path

          root = Path(os.environ['SOURCE_ROOT'])

          def replace(path: Path, old: str, new: str, label: str) -> None:
              body = path.read_text(encoding='utf-8')
              if old not in body:
                  raise SystemExit(f'Missing expected source for {label}: {path}')
              path.write_text(body.replace(old, new), encoding='utf-8')

          app_path = root / 'frontend' / 'app.json'
          app = json.loads(app_path.read_text(encoding='utf-8'))
          app['expo']['owner'] = os.environ['EAS_OWNER']
          app['expo'].setdefault('extra', {}).setdefault('eas', {})['projectId'] = os.environ['EAS_PROJECT_ID']
          app_path.write_text(json.dumps(app, indent=2) + '\n', encoding='utf-8')

          replace(
              root / 'scripts' / 'audit.py',
              'check(app_config.get("owner") == "wscott2k26", "Expo owner matches the canonical account")',
              'check(app_config.get("owner") == "wscott2k8", "Expo owner matches the connected EAS account")',
              'Expo audit identity',
          )

          replace(
              root / 'scripts' / 'audit-visual-master.py',
              'for source in sorted(FRONTEND.rglob("*.tsx")):\n    body = text(source)',
              'for source in sorted(FRONTEND.rglob("*.tsx")):\n    if "node_modules" in source.parts:\n        continue\n    body = text(source)',
              'visual audit scope',
          )

          replace(
              root / 'frontend' / 'app' / '(tabs)' / 'quiz.tsx',
              "icon: 'mount'",
              "icon: 'map'",
              'valid Ionicon name',
          )

          replace(
              root / 'frontend' / 'src' / 'components' / 'premium' / 'GlassPanel.tsx',
              '<Image pointerEvents="none" source={CAUSTICS} resizeMode="cover" style={styles.caustics} />',
              '<Image source={CAUSTICS} resizeMode="cover" style={styles.caustics} />',
              'GlassPanel Image props',
          )

          replace(
              root / 'frontend' / 'src' / 'components' / 'premium' / 'MaterialSurface.tsx',
              '<Image source={TEXTURES[material]} resizeMode="repeat" pointerEvents="none" style={styles.texture} />',
              '<Image source={TEXTURES[material]} resizeMode="repeat" style={styles.texture} />',
              'MaterialSurface Image props',
          )

          old_tab_icon = '''function tabIcon(name: keyof typeof Ionicons.glyphMap) {
  return ({ color, size, focused }: { color: string; size: number; focused: boolean }) => (
    <View style={[styles.iconShell, focused && styles.iconShellFocused]}>
      <Ionicons name={name} size={focused ? size + 1 : size} color={color} />
      {focused ? <View style={styles.activeDot} /> : null}
    </View>
  );
}'''
          new_tab_icon = '''function tabIcon(name: keyof typeof Ionicons.glyphMap) {
  const TabIcon = ({ color, size, focused }: { color: string; size: number; focused: boolean }) => (
    <View style={[styles.iconShell, focused && styles.iconShellFocused]}>
      <Ionicons name={name} size={focused ? size + 1 : size} color={color} />
      {focused ? <View style={styles.activeDot} /> : null}
    </View>
  );
  TabIcon.displayName = `TabIcon(${name})`;
  return TabIcon;
}'''
          replace(
              root / 'frontend' / 'app' / '(tabs)' / '_layout.tsx',
              old_tab_icon,
              new_tab_icon,
              'tab icon display name',
          )

          replace(
              root / 'frontend' / 'app' / '(tabs)' / 'journey.tsx',
              "import { colors, radii, spacing } from '@/src/theme';",
              "import { colors, spacing } from '@/src/theme';",
              'journey unused import',
          )

          replace(
              root / 'frontend' / 'app' / 'onboarding.tsx',
              "import { AVATARS, colors, radii, spacing } from '@/src/theme';",
              "import { AVATARS, colors } from '@/src/theme';",
              'onboarding unused imports',
          )

          root_layout = root / 'frontend' / 'app' / '_layout.tsx'
          replace(
              root_layout,
              "import { ErrorBoundary as ExpoRouterErrorBoundary, Stack } from 'expo-router';\n\nexport const ErrorBoundary = ExpoRouterErrorBoundary;\n\nimport * as SplashScreen from 'expo-splash-screen';",
              "import { ErrorBoundary as ExpoRouterErrorBoundary, Stack } from 'expo-router';\nimport * as SplashScreen from 'expo-splash-screen';",
              'root layout import order',
          )
          replace(
              root_layout,
              "import { colors } from '@/src/theme';\n\nvoid SplashScreen.preventAutoHideAsync();",
              "import { colors } from '@/src/theme';\n\nexport const ErrorBoundary = ExpoRouterErrorBoundary;\n\nvoid SplashScreen.preventAutoHideAsync();",
              'root layout ErrorBoundary placement',
          )

          puzzle_path = root / 'frontend' / 'app' / 'puzzle.tsx'
          replace(
              puzzle_path,
              '  useEffect(() => {\n    let active = true;',
              '  useEffect(() => {\n    let active = true;\n    const activeTimers = timers.current;',
              'puzzle timer snapshot',
          )
          replace(
              puzzle_path,
              '      timers.current.forEach(clearTimeout);',
              '      activeTimers.forEach(clearTimeout);',
              'puzzle timer cleanup',
          )

          replace(
              root / 'frontend' / 'src' / 'components' / 'premium' / 'TactileButton.tsx',
              "import { Pressable, StyleSheet, Text, View, type ViewStyle, type StyleProp } from 'react-native';",
              "import { Pressable, StyleSheet, Text, type ViewStyle, type StyleProp } from 'react-native';",
              'TactileButton unused import',
          )
          PY

      - name: Install registry-backed dependencies
        shell: bash
        run: |
          set -euo pipefail
          python -m pip install --disable-pip-version-check pillow==11.3.0
          npm install --global yarn@1.22.22
          cd "$SOURCE_ROOT/frontend"
          yarn install --non-interactive --network-timeout 600000

      - name: Verify deterministic artwork
        shell: bash
        run: |
          set -euo pipefail
          cd "$SOURCE_ROOT"
          find frontend/assets -type f -print0 | sort -z | xargs -0 sha256sum > /tmp/assets-before
          python scripts/generate-rc3-assets.py
          find frontend/assets -type f -print0 | sort -z | xargs -0 sha256sum > /tmp/assets-after
          diff -u /tmp/assets-before /tmp/assets-after

      - name: Run the full release suite four times
        shell: bash
        run: |
          set -euo pipefail
          cd "$SOURCE_ROOT"
          for pass in 1 2 3 4; do
            echo "========== RC3 CERTIFICATION PASS $pass OF 4 =========="
            python scripts/audit.py
            python scripts/audit-visual-master.py
            cd frontend
            yarn doctor
            yarn typecheck
            yarn lint
            rm -rf dist-ios dist-android .expo
            yarn export:ios
            yarn export:android
            test -d dist-ios && test -d dist-android
            rm -rf dist-ios dist-android .expo
            cd ..
          done

      - name: Create final manifest and publish normal source to main
        shell: bash
        run: |
          set -euo pipefail
          python - <<'PY'
          import hashlib, json, os
          from datetime import datetime, timezone
          from pathlib import Path
          root = Path(os.environ['SOURCE_ROOT'])
          ignored = {'.git', '.github', 'node_modules', '.expo', 'dist-ios', 'dist-android'}
          files = {}
          for path in sorted(root.rglob('*')):
              if not path.is_file() or path.name == 'BUILD_MANIFEST.json':
                  continue
              rel = path.relative_to(root)
              if any(part in ignored for part in rel.parts):
                  continue
              data = path.read_bytes()
              files[str(rel).replace('\\', '/')] = {'size': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
          manifest = {
              'generatedAt': datetime.now(timezone.utc).isoformat(),
              'sourceZipSha256': os.environ['RC3_SHA256'],
              'files': files,
          }
          (root / 'BUILD_MANIFEST.json').write_text(json.dumps(manifest, indent=2) + '\n', encoding='utf-8')
          PY

          rm -rf "$SOURCE_ROOT/frontend/node_modules" "$SOURCE_ROOT/frontend/.expo" "$SOURCE_ROOT/frontend/dist-ios" "$SOURCE_ROOT/frontend/dist-android"
          find . -mindepth 1 -maxdepth 1 ! -name '.git' ! -name '.github' ! -name 'rc3-upload-source' -exec rm -rf {} +
          rsync -a --exclude='.github/' "$SOURCE_ROOT/" ./
          rm -rf rc3-upload-source ci-payload payload release-patches

          git config user.email actions@users.noreply.github.com
          git config user.name 'Scripture Games CI'
          git add -A
          git commit -m 'Publish warning-free four-pass certified RC3 source'
          git push origin HEAD:main

          gh issue close 3 --repo "$GITHUB_REPOSITORY" --reason completed || true
          gh issue close 5 --repo "$GITHUB_REPOSITORY" --reason completed || true
          gh issue close 6 --repo "$GITHUB_REPOSITORY" --reason completed || true

      - name: Report a certification blocker
        if: failure()
        shell: bash
        run: |
          gh issue create --repo "$GITHUB_REPOSITORY" --title "RC3 V3 certification failed — run $GITHUB_RUN_ID" --body "Review: $GITHUB_SERVER_URL/$GITHUB_REPOSITORY/actions/runs/$GITHUB_RUN_ID"
